cmake_minimum_required(VERSION 3.20)
project(ReflexNative VERSION 0.6.7 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(reflex_engine STATIC src/engine.cpp src/runtime.cpp)
set_target_properties(reflex_engine PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(reflex_engine PUBLIC include)
if(MSVC)
  target_compile_options(reflex_engine PRIVATE /W4 /WX /EHsc)
else()
  target_compile_options(reflex_engine PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
option(REFLEX_BUILD_ADDON "Build the UXP Hybrid addon (requires Adobe SDK)" OFF)
set(REFLEX_UXP_SDK "${CMAKE_CURRENT_SOURCE_DIR}/sdk/uxp-hybrid-plugin-sdk-main" CACHE PATH "Adobe UXP Hybrid SDK folder")
if(REFLEX_BUILD_ADDON)
  foreach(header UxpAddonShared.h UxpAddonTypes.h)
    if(NOT EXISTS "${REFLEX_UXP_SDK}/src/api/${header}")
      message(FATAL_ERROR "Adobe UXP Hybrid SDK missing: ${header}. Set REFLEX_UXP_SDK; see native/README.md.")
    endif()
  endforeach()
  if(WIN32)
    set(REFLEX_PLATFORM src/platform_win.cpp)
  elseif(APPLE)
    enable_language(OBJCXX)
    set(REFLEX_PLATFORM src/platform_mac.mm)
  else()
    message(FATAL_ERROR "Photoshop addons support Windows and macOS only.")
  endif()
  add_library(reflex_addon SHARED src/addon.cpp ${REFLEX_PLATFORM}
    "${REFLEX_UXP_SDK}/src/utilities/UxpAddon.cpp"
    "${REFLEX_UXP_SDK}/src/utilities/UxpValue.cpp")
  target_include_directories(reflex_addon PRIVATE "${REFLEX_UXP_SDK}/src/utilities")
  target_link_libraries(reflex_addon PRIVATE reflex_engine)
  set_target_properties(reflex_addon PROPERTIES PREFIX "" OUTPUT_NAME "reflex" SUFFIX ".uxpaddon" OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED ON)
  if(WIN32)
    target_compile_definitions(reflex_addon PRIVATE UNICODE _UNICODE)
    target_link_libraries(reflex_addon PRIVATE gdiplus user32 gdi32)
  else()
    target_compile_options(reflex_addon PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
    target_link_libraries(reflex_addon PRIVATE "-framework AppKit" "-framework Carbon")
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(reflex_engine_tests tests/engine_tests.cpp)
  target_link_libraries(reflex_engine_tests PRIVATE reflex_engine)
  add_test(NAME reflex_engine COMMAND reflex_engine_tests)
  add_executable(reflex_runtime_tests tests/runtime_tests.cpp)
  target_link_libraries(reflex_runtime_tests PRIVATE reflex_engine)
  add_test(NAME reflex_runtime COMMAND reflex_runtime_tests)
  add_executable(reflex_ping_tests tests/ping_tests.cpp)
  target_link_libraries(reflex_ping_tests PRIVATE reflex_engine)
  add_test(NAME reflex_ping COMMAND reflex_ping_tests)
  if(WIN32)
    add_executable(reflex_windows_paint_tests tests/windows_paint_tests.cpp)
    target_compile_definitions(reflex_windows_paint_tests PRIVATE UNICODE _UNICODE)
    target_link_libraries(reflex_windows_paint_tests PRIVATE reflex_engine gdiplus user32 gdi32)
    add_test(NAME reflex_windows_paint COMMAND reflex_windows_paint_tests)
    add_executable(reflex_windows_overlay_tests tests/windows_overlay_tests.cpp)
    target_include_directories(reflex_windows_overlay_tests PRIVATE include)
    target_compile_definitions(reflex_windows_overlay_tests PRIVATE UNICODE _UNICODE)
    target_link_libraries(reflex_windows_overlay_tests PRIVATE user32)
    add_test(NAME reflex_windows_overlay COMMAND reflex_windows_overlay_tests)
    set_tests_properties(reflex_windows_overlay PROPERTIES TIMEOUT 15)
  endif()
  if(REFLEX_BUILD_ADDON)
    add_executable(reflex_addon_tests tests/addon_tests.cpp src/addon.cpp
      "${REFLEX_UXP_SDK}/src/utilities/UxpAddon.cpp"
      "${REFLEX_UXP_SDK}/src/utilities/UxpValue.cpp")
    target_include_directories(reflex_addon_tests PRIVATE "${REFLEX_UXP_SDK}/src/utilities")
    target_link_libraries(reflex_addon_tests PRIVATE reflex_engine)
    add_test(NAME reflex_addon COMMAND reflex_addon_tests)
  endif()
endif()
